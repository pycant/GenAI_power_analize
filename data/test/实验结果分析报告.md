def fibonacci(n):
  """
  计算斐波那契数列的第n项。

  Args:
    n: 一个非负整数，表示要计算的斐波那契数列的项数。

  Returns:
    斐波那契数列的第n项。
  """

  if n <= 0:
    return 0
  elif n == 1:
    return 1
  else:
    a, b = 0, 1
    for _ in range(2, n + 1):
      a, b = b, a + b
    return b

# 示例用法
print(fibonacci(0))  # 输出: 0
print(fibonacci(1))  # 输出: 1
print(fibonacci(2))  # 输出: 1
